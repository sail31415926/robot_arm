# robot_arm 包族 Python 依赖 —— 核心组（必需）
# ---------------------------------------------------------------------------
# 这些库叠加在 ROS 2 Humble 的系统 site-packages 之上（venv 用
# --system-site-packages 创建）。ROS 相关包不在此处，由 apt + source ROS 提供：
#   rclpy / sensor_msgs / geometry_msgs / trajectory_msgs / control_msgs /
#   diagnostic_msgs / moveit_msgs / cv_bridge / tf2_ros / tf_transformations /
#   message_filters / launch / launch_ros / ament_index_python ...
# tkinter 也走 apt（python3-tk），不在此处。
#
# 本文件只放**跑仿真 / 实机 / 调试 GUI 必需**的包。可选功能（MuJoCo、RL 训练、
# MATLAB 离线分析）的依赖在 requirements-optional.txt，按组用 setup_venv.sh 的
# --with-mujoco / --with-rl / --with-analysis / --all 安装。
#
# 约束：numpy 必须 <2（否则 cv2 等 ABI 冲突）。
# 版本与当前已验证的 .venv 实测一致，便于复现。
#
# 一键安装（推荐）：bash setup_venv.sh
# 手工安装：       pip install -r requirements.txt

numpy==1.26.4            # 全部子包；必须 <2，否则 cv2 等 ABI 冲突
PyYAML==5.4.1            # 配置读取（launch / 控制器参数）
                         # 由 apt python3-yaml 提供，pip 只做版本校验、不重新构建
                         # （5.4.1 源码与 Cython 3 不兼容，强制从源码装会失败）
ruckig==0.17.3           # 在线轨迹生成（jerk-limited OTG；运镜 / 点到点）
PyQt5==5.15.11           # joint_position_gui 等滑块 GUI
                         # （tkinter GUI 由 apt python3-tk 提供，无需 pip）

# ---------------------------------------------------------------------------
# cv2 不用 pip 装 —— 由 apt python3-opencv 提供（当前实测 4.5.4）
# ---------------------------------------------------------------------------
# 两条硬理由，改之前务必先读：
#
# ① 不能装 opencv-python-headless：headless wheel 是**不带 GUI 支持**编译的，
#    而 robot_arm_debug/python/vision/red_box_detector.py 默认 show_window:=true，
#    会调 cv2.namedWindow / imshow / waitKey → 直接抛
#    "The function is not implemented. Rebuild the library with ... support"。
#
# ② 不能装 4.12+ 的任何 opencv-python*：它们依赖 numpy>=2，与上面的 numpy<2
#    硬约束直接冲突，pip 会报 ResolutionImpossible（旧版本本文件曾 pin
#    opencv-python-headless==4.13.0.92，在全新机器上根本装不上；老环境里
#    只留下残缺的 dist-info，实际 import 到的一直是 apt 那个版本）。
#
# 如果确实需要比 apt 更新的 cv2，装**完整版**（带 GUI）且不越过 numpy<2 这道线：
#    pip install "opencv-python==4.11.0.86"      # 4.11.0.86 是最后一个兼容 numpy 1.26 的
# 装完用 setup_venv.sh --check 确认 cv2 的来源与 GUI 支持都正常。
